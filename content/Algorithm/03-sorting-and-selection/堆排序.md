标准思路：

1. 自底向上建立大根堆；
2. 每次把堆顶最大值交换到当前无序区的末尾；
3. 缩小堆范围；
4. 对新的根结点执行下沉调整；
5. 最终得到递增数组。

```java
class Solution {
public:
    vector<int> sortArray(vector<int>& nums) {
        int heapSize = static_cast<int>(nums.size());

        buildMaxHeap(nums, heapSize);

        // 每轮将当前最大值放入已排序区的最前端
        for (int end = heapSize - 1; end > 0; --end) {
            swap(nums[0], nums[end]);

            // [0, end) 仍是待维护的大根堆
            siftDown(nums, 0, end);
        }

        return nums;
    }

private:
    // 从最后一个非叶子结点开始，自底向上执行向下调整。
    void buildMaxHeap(vector<int>& nums, int heapSize) {
        // heapSize 是元素个数，相当于从 1 开始给节点编号
        // heapSize / 2 - 1 计算的是最后一个非叶子节点的索引
        for (int i = heapSize / 2 - 1; i >= 0; --i) {
            siftDown(nums, i, heapSize);
        }
    }

    // 保证以 idx 为根的子树满足大根堆性质。
    // heapSize 表示堆区间为 [0, heapSize)。
    void siftDown(vector<int>& nums, int idx, int heapSize) {
        int largest = idx;
        int left = idx * 2 + 1;
        int right = idx * 2 + 2;

        if (left < heapSize && nums[left] > nums[largest]) {
            largest = left;
        }

        if (right < heapSize && nums[right] > nums[largest]) {
            largest = right;
        }

        if (largest != idx) {
            swap(nums[idx], nums[largest]);
            siftDown(nums, largest, heapSize);
        }
    }
};
```
