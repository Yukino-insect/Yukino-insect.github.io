归并也是一种基于分值思想的排序算法。程序通过递归，不断的将数组或者其子数组分成两部分。当子数组的元素数为 1，即 `left >= right` 时，子数组中只剩下一个元素，天然后续。然后程序执行合并操作，将两个元素合并成一个有序的子数组。不断执行上述过程，即可得到有序序列。

```c++
class Solution {
public:
    vector<int> sortArray(vector<int>& nums) {
        vector<int> temp(nums.size()); // 用于辅助排序
        mergeSort(nums, 0, nums.size() - 1, temp);
        return nums;
    }

    void mergeSort(vector<int>& nums, int left, int right, vector<int>& temp) {
        if (left >= right) return;
        int mid = (left + right) / 2; // 将数组划分成两部分
        mergeSort(nums, left, mid, temp); // 对左子数组进行递归
        mergeSort(nums, mid + 1, right, temp); // 对有子数组进行递归
        // 该操作为合并操作。
        // 代码会先将左子数组递归完成并排序成有序数组
        // 然后会对右子数组递归完成并排序为有序数组
        // 整个递归就像一个树一样
        merge(nums, left, mid, right, temp);
    }

    void merge(vector<int>& nums, int left, int mid, int right, vector<int>& temp) {
        int i = left, j = mid + 1; // 左右子数组的起点元素
        int k = left; // temp 待插入位置

        // 将左右子数组合并成一个数组
        // 注意下列循环中的终止条件，存在两个边界问题
        while (i <= mid && j <= right) {
            if (nums[i] <= nums[j]) {
                temp[k++] = nums[i++];
            } else {
                temp[k++] = nums[j++];
            }
        }

        if (i <= mid) {
            while (i <= mid) temp[k++] = nums[i++];
        } else {
            while (j <= right) temp[k++] = nums[j++];
        }

        while (left <= right) {
            nums[left] = temp[left];
            left++;
        }
    }
};
```
