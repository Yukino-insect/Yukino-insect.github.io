快速排序采用归并的思想。

找到一个 `pivot` 做基准值。将数组分成两部分，如果是单调不减排序，左半部分全都是小于等于 `pivot` 的值，右半部分全都是大于等于 `pivot` 的值。然后不断对左右两部分数组进行递归操作。当子递归函数中的 `left` > `right`，递归结束。代码如下：
```c++
class Solution {
public:
    vector<int> sortArray(vector<int>& nums) {
        quickSort(nums, 0, nums.size() - 1);
        return nums;
    }

    void quickSort(vector<int>& nums, int left, int right) {
        if (left >= right) return;
        int pivot = partition(nums, left, right);
        quickSort(nums, left, pivot - 1);
        quickSort(nums, pivot + 1, right);
    }

    int partition(vector<int>& nums, int left, int right) {
        int i = left, j = right;
        while (i < j) {
            // 从右向左遍历
            // 必须使用非严格不等，如果使用严格不等
            while (i < j && nums[j] >= nums[left]) j--; // 遇到小于 pivot 的元素停止
            while (i < j && nums[i] <= nums[left]) i++; // 遇到大于 pivot 的元素停止
            swap(nums[i], nums[j]); // 交换
        } 
        swap(nums[left], nums[i]); // 因为 i 是后执行的，当 i >= j 时，i 处的值一定小于等于 pivot
        return i;
    }

    int medianOfThree(vector<int>& nums, int left, int mid, int right) {
        int a = nums[left];
        int b = nums[mid];
        int c = nums[right];

        if ((a <= b && b <= c) || (c <= b && b <= a)) {
            return mid;
        }

        if ((b <= a && a <= c) || (c <= a && a <= b)) {
            return left;
        }

        return right;
    }
};
```

上述代码在[力扣数组排序](https://leetcode.cn/problems/sort-an-array/)中，上述代码会超时。但是如果我们修改一下基准的选择就可以通过，代码如下：
```c++
class Solution {
public:
    vector<int> sortArray(vector<int>& nums) {
        quickSort(nums, 0, nums.size() - 1);
        return nums;
    }

    void quickSort(vector<int>& nums, int left, int right) {
        if (left >= right) return;
        int pivot = partition(nums, left, right);
        quickSort(nums, left, pivot - 1);
        quickSort(nums, pivot + 1, right);
    }

    int partition(vector<int>& nums, int left, int right) {
        int mid = medianOfThree(nums, left, (left + right) / 2, right);
        swap(nums[left], nums[mid]);
        int i = left, j = right;
        while (i < j) {
            // 从右向左遍历
            // 必须使用非严格不等，如果使用严格不等
            while (i < j && nums[j] >= nums[left]) j--; // 遇到小于 pivot 的元素停止
            while (i < j && nums[i] <= nums[left]) i++; // 遇到大于 pivot 的元素停止
            swap(nums[i], nums[j]); // 交换
        } 
        swap(nums[left], nums[i]); // 因为 i 是后执行的，当 i >= j 时，i 处的值一定小于等于 pivot
        return i;
    }

    int medianOfThree(vector<int>& nums, int left, int mid, int right) {
        int a = nums[left];
        int b = nums[mid];
        int c = nums[right];

        if ((a <= b && b <= c) || (c <= b && b <= a)) {
            return mid;
        }

        if ((b <= a && a <= c) || (c <= a && a <= b)) {
            return left;
        }

        return right;
    }
};
```

原来我们直接使用数组的最左侧的值作为基准，如果排序数组有序部分较多，算法会发生退化，因此我们可以选择一个中位数来优化该情况。

如果选择的基准在数组中存在大量相同元素，这样也会对算法的效率产生影响，因此我们可以写下三路排序，将相同的基准元素排除出去。代码如下：

```c++
class Solution {
public:
    vector<int> sortArray(vector<int>& nums) {
        quickSort(nums, 0, nums.size() - 1);
        return nums;
    }

    void quickSort(vector<int>& nums, int left, int right) {
        if (left >= right) return;
        
        swap(nums[left], nums[medianOfThree(nums, left, (left + right) / 2, right)]);
        int pivot = nums[left]; // 基准值
        int lt = left; // 指向小于部分待插入的位置
        int gt = right; // 指向大于部分待插入的位置
        int i = left + 1; // 遍历整个数组的指针

        while (i <= gt) { // 遇到 gt 并检查后，即可停止循环
            if (nums[i] < pivot) {
                swap(nums[i], nums[lt]);
                lt++;
                i++;
            } else if (nums[i] > pivot) {
                swap(nums[i], nums[gt]);
                gt--; // 这里不需要 i++，因为交换到 i 位置的 gt 指的元素没有被检查过。
            } else {
                i++;
            }
        }
        quickSort(nums, left, lt - 1);
        quickSort(nums, gt + 1, right);
    }

    int medianOfThree(vector<int>& nums, int left, int mid, int right) {
        int a = nums[left];
        int b = nums[mid];
        int c = nums[right];

        if ((a <= b && b <= c) || (c <= b && b <= a)) {
            return mid;
        }

        if ((b <= a && a <= c) || (c <= a && a <= b)) {
            return left;
        }

        return right;
    }
};
```

上述代码通过 `i` 指针遍历整个数组，由于 `left` 位置是基准元素，因此从 `left + 1` 开始遍历。在遍历的过程中，`i` 指向的元素如果小于基准值，就将其放到 `lt` 位置 `lt++` 更新位置，`lt` 更新后的位置是未知的，因此小于 `pivot` 数组的部分是 `[left, lt - 1]`，当前采用的是闭区间，`gt` 同理。如果 `i` 第一次遇到等于基准的元素，`i++`，不发生交换，当下一次满足条件发生交换时，等于基准的元素就会被换到 `i` 的位置，最终等于基准的元素会集中到 `[lt, gt]` 区间。再次递归时不会操作这部分数据。 

上述代码 `partition` 方法也可以这样实现：
```c++
private int partition(vector<int>& nums, int left, int right) {
    int pivot = left; // 基准元素
    int le = pos + 1; // 小于 pivot 的待插入位置

    for (int i = le; i <= right; i++) {
        if (nums[i] < nums[pivot]) {
            swap(nums[i], nums[le]);
            le++;
        }
    }
	// 最终所有大于等于 pivot 的元素都会被交换到 [le, right] 区间
    // 小于 pivot 的元素都会被交换到 [left + 1, le - 1] 的区间
    // 道理很好理解，i 遍历了整个数组。le 在插入数据后会立即 le++，因此 le 会始终指向一个待插入位置
    // 最后将 pivot 上的基准元素交换到 le - 1
    swap(nums[pivot], nums[le - 1]);

    return le - 1;
}
```
